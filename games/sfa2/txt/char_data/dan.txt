-- Basic states
	-- Idle & movement
0x1a20 - 0x1a98: Idle
0x1ab4: Crouching
0x1ad0 - 0x1b48: Forward walk
0x1b64 - 0x1bdc: Back walk

0x1cc0 - 0x1d08: Standing turn around after side switch
0x1d24 - 0x1d6c: Crouching turn around after side switch
0x1d88 - 0x1da0: Prejump frames
0x1dbc - 0x1dd4: Landing recovery from jump
0x1df0 - 0x1e68: Neutral jump

0x1f18 - 0x1f90: Backward jump
