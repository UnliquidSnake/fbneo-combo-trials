-- Basic states
	-- Idle & movement
0x0ca6 - 0x0d1e: Idle
0x0d3a: Crouching
0x0d56 - 0x0dce: Forward walk
0x0dea - 0x0e62: Back walk
0x0e7e - 0x0ec6: Transition from crouch to stand
0x0ee2 - 0x0f2a: Transition from stand to crouch
0x0f46 - 0x0f8e: Standing turn around after side switch
0x0faa - 0x0ff2: Crouching turn around after side switch
0x100e - 0x1026: Prejump frames
0x1042 - 0x105a: Landing recovery from jump
0x1076 - 0x1106: Neutral jump
0x1122 - 0x11ca: Forward jump
0x11e6 - 0x128e: Backward jump
	-- Blocking
0x12aa - 0x130e: Stand block (incl. proximity block)
0x132a - 0x138e: Crouch block (incl. proximity block)
0x13c2 - 0x1426: Air block (incl. proximity block)
	-- Recovery Roll
0x14a6 - 0x14ee: Auto roll wakeup
0x150a - 0x1552: Recovery roll
	-- Custom Combo
0x6f36 - 0x6f4e: Grounded CC activation
0x6f6a - 0x6f82: Air CC activation
